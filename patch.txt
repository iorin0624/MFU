<script>
  // === 役割に応じてメモ欄（costume_label）を有効/無効化 ===
  (function(){
    function isMemoEnabled(root){
      var checked = root.querySelector('input[name="participant_role"]:checked');
      if(!checked) return false;
      return (checked.value === 'cosplayer' || checked.value === 'other');
    }
    function setMemoState(root){
      var input = root.querySelector('.js-costume-input');
      if(!input) return;
      if(isMemoEnabled(root)){
        input.removeAttribute('disabled');
        if(!input.placeholder || input.placeholder.indexOf('衣装') !== -1){
          input.placeholder = '衣装／その他のメモ';
        }
      }else{
        if(input.value) input.value = '';
        input.setAttribute('disabled','disabled');
        input.placeholder = '衣装／その他（該当時のみ入力可）';
      }
    }
    var form = document.querySelector('.js-bulk-form');
    if(form){
      form.addEventListener('change', function(e){
        if(e.target && e.target.name === 'participant_role'){
          setMemoState(form);
        }
      });
      form.addEventListener('submit', function(){ setMemoState(form); });
      setMemoState(form);
    }
  })();
</script>
{% endblock %}
