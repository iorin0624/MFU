 def fmt_yen(value, digits=0):
     if value is None or value == "":
         return "-"
     try:
         number = float(value)
     except Exception:
         return str(value)
     precision = int(digits)
     if precision > 0:
         return f"{round(number, precision):,.{precision}f}"
     return f"{round(number):,}"
