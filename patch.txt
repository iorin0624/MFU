+  function handleMentionInput() {
+    if (!bodyInput) return;
+    const caret = Number(bodyInput.selectionStart || 0);
+    const text = bodyInput.value || '';
+    const head = text.slice(0, caret);
+    const at = head.lastIndexOf('@');

+      closeMentionMenu();
+      return;
+    }

+    if (!prefix || /[\s.,!?()\[\]{}]/.test(prefix)) {
+      closeMentionMenu();
+      return;
+    }
+    state.mentionRange = { start: at, end: caret };
+    if (state.mentionDebounceTimer) clearTimeout(state.mentionDebounceTimer);
+    state.mentionDebounceTimer = setTimeout(() => {
+      fetchMentionCandidates(prefix);
+    }, 200);
+  }

+  function handleMentionKeydown(e) {
+    if (!state.mentionCandidates.length) return;
+    if (e.key === 'ArrowDown') {
+      e.preventDefault();
+      state.mentionActiveIndex = (state.mentionActiveIndex + 1) % state.mentionCandidates.length;
+      renderMentionMenu();
+    } else if (e.key === 'ArrowUp') {
+      e.preventDefault();
+      state.mentionActiveIndex = (state.mentionActiveIndex - 1 + state.mentionCandidates.length) % state.mentionCandidates.length;
+      renderMentionMenu();
+    } else if (e.key === 'Enter') {
+      e.preventDefault();
+      applyMentionCandidate(state.mentionActiveIndex);
+    } else if (e.key === 'Escape') {
+      e.preventDefault();
+      closeMentionMenu();
+    }
+  }

+    if (!candidate || !state.mentionRange || !bodyInput) return;
