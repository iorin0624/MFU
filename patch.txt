+def _normalize_multiline_list(lines):
+    normalized = []
+    seen = set()
+    for line in lines or []:
+        item = (line or "").strip()
+        if not item:
+            continue
+        dedupe_key = item.lower()
+        if dedupe_key in seen:
+            continue
+        seen.add(dedupe_key)
+        normalized.append(item)
+    return normalized
+
