 def _parse_payload_datetime(value: Any) -> datetime | None:
     if not value or not isinstance(value, str):
         return None
     text = value.strip()
     patterns = ["%Y-%m-%d %H:%M:%S", "%Y-%m-%d %H:%M", "%Y-%m-%d"]
     for pattern in patterns:
         try:
             return datetime.strptime(text, pattern)
         except ValueError:
             continue
     return None
