 def update_invoice_status(invoice_id: int, status: str) -> None:
     status = normalize_status(status)
     now = now_jst()
     db = get_db()
     cur = db.cursor()
     try:
         cur.execute(
             "UPDATE invoice_headers SET status=%s, updated_at=%s WHERE id=%s",
             (status, now, invoice_id),
         )
         db.commit()
     finally:
         cur.close()
         db.close()
