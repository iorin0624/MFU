   const fallbackPath = eventId ? `/chat/events/${eventId}` : '/';
   const candidate = typeof rawUrl === 'string' && rawUrl.trim() ? rawUrl.trim() : fallbackPath;
 
   if (candidate.startsWith('//')) {
     return `${self.location.origin}/`;
   }
