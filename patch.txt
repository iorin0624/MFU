 def _redirect_login():
     path = request.full_path or request.path or "/"
     if request.path.startswith("/external-login/admin"):
         return redirect(url_for("login", next=path))
     if request.path.startswith("/external-login"):
         return redirect(url_for("external_login_user.line_login", next=path))
     return redirect(url_for("login", next=path))
 
 
 def user_has_feature(user_id: str, feature_key: str) -> bool:
     if not user_id or not feature_key:
         return False
     if user_id == "admin":
         return True
