 def _is_same_origin_request():
     expected = urlparse(request.host_url)
     for header_name in ("Origin", "Referer"):
         raw = request.headers.get(header_name)
         if not raw:
             continue
         parsed = urlparse(raw)
         if parsed.scheme != expected.scheme or parsed.netloc != expected.netloc:
             return False
     return True
