 def _uuid_bytes_to_str(b: bytes | None) -> Optional[str]:
     if not b: return None
     try:
         return str(uuid.UUID(bytes=b))
     except Exception:
         try:
             return str(uuid.UUID(hex=b.hex()))
         except Exception:
             return None
