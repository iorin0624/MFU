   async function copyTextWithFallback(text) {
     if (!text) return;
     if (navigator.clipboard?.writeText) {
       await navigator.clipboard.writeText(text);
       return;
     }
     const temp = document.createElement('textarea');
     temp.value = text;
     temp.setAttribute('readonly', 'readonly');
     temp.style.position = 'fixed';
     temp.style.left = '-9999px';
     document.body.appendChild(temp);
     temp.select();
     document.execCommand('copy');
     document.body.removeChild(temp);
   }

   function applyReadUpdate(readState) {
     if (!readState?.actor_type || !readState?.actor_id) return;
     const key = `${readState.actor_type}:${readState.actor_id}`;
     const prev = state.readStateByActor.get(key);
     const nextReadId = Number(readState.last_read_message_id || 0);
     const prevReadId = Number(prev?.last_read_message_id || 0);
     if (prev && prevReadId > nextReadId) return;
     state.readStateByActor.set(key, {
       actor_type: readState.actor_type,
       actor_id: String(readState.actor_id),
       display_name: readState.display_name || prev?.display_name || readState.actor_id,
       last_read_message_id: Math.max(prevReadId, nextReadId),
     });
