     """同名が存在したら (n) を付けて衝突回避。"""
     root, ext = os.path.splitext(filename)
     candidate = filename
     full = os.path.join(dst_dir, candidate)
     n = 1
     while os.path.exists(full):

         n += 1
     return full, candidate
