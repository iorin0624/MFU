 def _parse_actor_key(value: str) -> tuple[str, str] | None:
     raw = (value or "").strip()
     if ":" not in raw:
         return None
     actor_type, actor_id = raw.split(":", 1)
     actor_type = actor_type.strip()
     actor_id = actor_id.strip()
     if not actor_type or not actor_id:
         return None
     return actor_type, actor_id
